# QuestForge — Unity Engine Integration Guide

This guide explains how to import and parse the static JSON `game_package/` generated by QuestForge into a Unity point-and-click adventure game.

---

## 1. Package Directory Structure

```
game_package/
├── manifest.json              # Package metadata and entry chapter
├── world_bible.json           # Locations, factions, NPCs, rules, timeline
├── characters.json            # Character sheets and relationship graph
├── trait_arc.json             # Trait names, colors, and tier thresholds
├── chapters/
│   ├── chapter_01.json        # Contract, scenes, dialogue tree, and local quests
│   └── chapter_0N.json
├── side_quests/               # Optional and latent advantage quests
├── achievements.json          # Unlockable badges and trigger flags
└── attachment_points.json     # Side quest injection slots
```

## 2. Unity C# Data Models

Create `QuestForgeModels.cs` in your Unity project `Assets/Scripts/Narrative/`:

```csharp
using System;
using System.Collections.Generic;
using UnityEngine;

namespace QuestForge.Runtime
{
    [Serializable]
    public class ManifestData
    {
        public string title;
        public string version;
        public string schema_version;
        public int chapters_count;
        public List<string> traits;
        public int total_quests;
        public int total_achievements;
        public string entry_chapter_id;
    }

    [Serializable]
    public class DialogueChoiceData
    {
        public string label;
        public string trait_tag;
        public List<string> flags_set;
        public string next_node;
        public string condition_flag;
    }

    [Serializable]
    public class DialogueNodeData
    {
        public string node_id;
        public string type; // "speaker" | "choice" | "flag" | "end"
        public string speaker;
        public string text;
        public List<DialogueChoiceData> choices;
        public string next_node;
    }

    [Serializable]
    public class QuestData
    {
        public string quest_id;
        public string title;
        public string type; // "main_blocking" | "latent_advantage" | "neutral" | "achievement"
        public string chapter_id;
        public string location_id;
        public string trigger_flag;
        public string objective;
        public string journal_entry;
        public List<string> reward_flags;
    }

    [Serializable]
    public class ChapterData
    {
        public string chapter_id;
        public List<DialogueNodeData> dialogue_tree;
        public List<QuestData> quests;
        public List<string> characters_present;
    }
}
```

## 3. Unity Package Loader Script

Create `QuestForgeLoader.cs` in `Assets/Scripts/Narrative/`:

```csharp
using System.IO;
using UnityEngine;

namespace QuestForge.Runtime
{
    public class QuestForgeLoader : MonoBehaviour
    {
        [Header("Package Path (StreamingAssets)")]
        public string packageFolder = "game_package";

        public ManifestData Manifest { get; private set; }
        public ChapterData CurrentChapter { get; private set; }

        public void LoadPackage()
        {
            string basePath = Path.Combine(Application.streamingAssetsPath, packageFolder);

            // 1. Read Manifest
            string manifestPath = Path.Combine(basePath, "manifest.json");
            if (File.Exists(manifestPath))
            {
                string json = File.ReadAllText(manifestPath);
                Manifest = JsonUtility.FromJson<ManifestData>(json);
                Debug.Log($"[QuestForge] Loaded Story: {Manifest.title} ({Manifest.chapters_count} Chapters)");
            }

            // 2. Load Entry Chapter
            LoadChapter(Manifest.entry_chapter_id);
        }

        public void LoadChapter(string chapterId)
        {
            string chapterPath = Path.Combine(Application.streamingAssetsPath, packageFolder, "chapters", $"{chapterId}.json");
            if (File.Exists(chapterPath))
            {
                string json = File.ReadAllText(chapterPath);
                CurrentChapter = JsonUtility.FromJson<ChapterData>(json);
                Debug.Log($"[QuestForge] Active Chapter: {CurrentChapter.chapter_id} with {CurrentChapter.dialogue_tree.Count} dialogue nodes.");
            }
        }
    }
}
```

## 4. Runtime Trait Tracking in Unity

```csharp
public class TraitTracker : MonoBehaviour
{
    private Dictionary<string, int> traitCounts = new Dictionary<string, int>();

    public void OnChoiceSelected(DialogueChoiceData choice)
    {
        if (!string.IsNullOrEmpty(choice.trait_tag))
        {
            if (!traitCounts.ContainsKey(choice.trait_tag))
                traitCounts[choice.trait_tag] = 0;

            traitCounts[choice.trait_tag]++;
            Debug.Log($"[QuestForge] Trait '{choice.trait_tag}' increased to {traitCounts[choice.trait_tag]}");
        }
    }

    public string GetTraitTier(string traitName)
    {
        int count = traitCounts.ContainsKey(traitName) ? traitCounts[traitName] : 0;
        if (count >= 8) return "dominant";
        if (count >= 4) return "established";
        if (count >= 1) return "emerging";
        return "none";
    }
}
```
